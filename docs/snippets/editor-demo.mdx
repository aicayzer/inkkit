const full = ${fullPage};

const status = $('status');
const root = $('editor');
const initial = '# Design review\\n\\nWe tried the new note view with a few longer documents. The **quieter toolbar** leaves more room for writing.\\n\\n## What we learned\\n\\n- Keep the title visible when opening a note.\\n- Put less-used commands in the menu.\\n- Give attachments clear names.\\n\\n## Next steps\\n\\n- [x] Compare light and dark appearances\\n- [ ] Try a note with several tables\\n\\n| Area | Next step |\\n| --- | --- |\\n| Navigation | Check keyboard shortcuts |\\n| Writing | Review paste and undo |\\n\\n> Keep the writing easy to read, and leave room for the details.\\n\\nRead the [appearance guide](https://inkkit.cyzr.me/guides/appearance) to customise a preset. '\\n## A balanced week\\n\\n\`\`\`mermaid\\npie title A balanced week\\n  "Writing" : 35\\n  "Review" : 20\\n  "Planning" : 15\\n  "Research" : 12\\n  "Design" : 8\\n  "Support" : 5\\n  "Admin" : 3\\n  "Learning" : 2\\n\`\`\`\\n' : '');
const idle = '';
const say = (text, error = false) => { status.textContent = text; status.classList.toggle('status-error', error); };
const icons = ${JSON.stringify({ focus: icons.focus, unfocus: icons.unfocus })};
let editor;
let generation = 0;
let mode = 'formatted';
let focused = false;
let siteScheme = matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';
const applyAppearance = () => {

  const scheme = selected === 'system' ? siteScheme : selected;
  document.documentElement.style.colorScheme = scheme;
  root.style.setProperty('--inkkit-color-scheme', scheme);

  $('theme').onchange = () => { root.dataset.inkkitTheme = $('theme').value; };
  $('appearance').onchange = applyAppearance;

  const button = $('focus');

  const label = focused ? 'Exit focus' : 'Focus';
  button.innerHTML = icons[focused ? 'unfocus' : 'focus'];
  button.setAttribute('aria-label', label);
  button.title = focused ? 'Exit focus (Esc)' : 'Focus';

  mode = state.mode;
  if (full) {
    $('formatted').setAttribute('aria-pressed', String(mode === 'formatted'));
    $('source').setAttribute('aria-pressed', String(mode === 'source'));
    $('undo').disabled = !state.commands.undo;
    $('redo').disabled = !state.commands.redo;

      try { await navigator.clipboard.writeText(text); return true; } catch {}

      area.setAttribute('readonly', '');

    // ProseMirror always cancels Esc inside the text, so a cancelled Esc only counts as handled

    document.addEventListener('keydown', () => { slashOpen = root.querySelector('[role=combobox][aria-expanded=true]') !== null; }, true);

      if (event.key === 'Escape') {
        const handled = slashOpen || (event.defaultPrevented && !event.target.isContentEditable);
        if (focused && !handled && !event.isComposing) parent.postMessage({ type: 'inkkit-demo-focus', focused: false }, '*');
      } else if ((event.key === 'F3' || (event.key === 'g' && (event.metaKey || event.ctrlKey))) && query.value) { event.preventDefault(); findNext(); }
    });

      const keys = { ArrowLeft: -1, ArrowRight: 1, Home: 'first', End: 'last' };

      const items = [...bar.querySelectorAll('button:not(:disabled)')];
      const at = items.indexOf(event.target);
      const step = keys[event.key];
      const next = step === 'first' ? 0 : step === 'last' ? items.length - 1 : (at + step + items.length) % items.length;
      event.preventDefault();
      items[next].focus();
    });
  }
} catch (error) {

export const EditorPage = () => {
  const [focused, setFocused] = useState(false)
  useEffect(() => {
    document.documentElement.classList.toggle('editor-focus', focused)
    return () => document.documentElement.classList.remove('editor-focus')
  }, [focused])
  useEffect(() => {
    if (!focused) return
    const onKeyDown = (event) => {
      if (event.key === 'Escape' && !event.defaultPrevented) setFocused(false)
    }
    document.addEventListener('keydown', onKeyDown)
    return () => document.removeEventListener('keydown', onKeyDown)
  }, [focused])
  return (
    <div className="editor-page">
      <EditorDemo fullPage focused={focused} onFocusChange={setFocused} />
    </div>
  )
}
