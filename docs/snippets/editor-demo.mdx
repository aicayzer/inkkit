export const EditorDemo = ({ fullPage = false }) => {
  const frame = useRef(null);
  const sendTheme = () => {
    frame.current?.contentWindow?.postMessage(
      {
        type: "inkkit-demo-theme",
        dark: document.documentElement.classList.contains("dark"),
      },
      "*",
    );
  };
  useEffect(() => {
    const observer = new MutationObserver(sendTheme);
    observer.observe(document.documentElement, {
      attributes: true,
      attributeFilter: ["class"],
    });
    return () => observer.disconnect();
  }, []);
  const html = `<!doctype html>
<html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1">
<link rel="stylesheet" href="https://cdn.jsdelivr.net/npm/@aicayzer/inkkit@0.1.7/dist/style.css">
<style>
:root { color-scheme: light dark; }
* { box-sizing: border-box; }
body { margin: 0; font: 13px/1.5 system-ui,sans-serif; color: CanvasText; background: light-dark(#fff,#0b0b0b); }
header { display: flex; align-items: center; justify-content: space-between; gap: 12px; padding: 10px 20px; border-bottom: 1px solid light-dark(#e5e5e5,#292929); }
header span { opacity: .6; }
nav { display: flex; gap: 4px; }
button { font: inherit; color: inherit; border: 0; background: transparent; border-radius: 5px; padding: 4px 8px; cursor: pointer; }
button:hover,button[aria-pressed=true] { background: light-dark(#f0f0f0,#252525); }
button:disabled { opacity: .4; cursor: default; }
#editor { --inkkit-font-family: system-ui,sans-serif; --inkkit-font-size: 15px; --inkkit-line-height: 1.55; --inkkit-reading-width: ${fullPage ? "48rem" : "100%"}; --inkkit-padding-inline: 24px; --inkkit-surface-focus-width: 0; min-height: 410px; padding-block: 18px; }
#status { font-size: 12px; opacity: .65; padding: 0 24px 12px; margin: 0; }
@media(max-width:440px) { header { padding-inline: 12px; } header span { max-width: 110px; } #editor { --inkkit-padding-inline: 16px; } }
</style></head><body>
<header><span>Try editing this note</span><nav aria-label="Demo controls"><button id="mode" disabled aria-pressed="false">Source</button><button id="reset" disabled>Reset</button></nav></header>
<div id="editor"></div><p id="status" role="status">Loading editor…</p>
<script type="module">
const status = document.getElementById('status');
const root = document.getElementById('editor');
const mode = document.getElementById('mode');
const reset = document.getElementById('reset');
const initial = '# Design review\\n\\nWe tried the new note view with a few longer documents. The **quieter toolbar** leaves more room for writing.\\n\\n## What we learned\\n\\n- Keep the title visible when opening a note.\\n- Put less-used commands in the menu.\\n- Give attachments clear names.\\n\\n## Next steps\\n\\n- [x] Compare light and dark appearances\\n- [ ] Try a note with several tables\\n\\n| Area | Next step |\\n| --- | --- |\\n| Navigation | Check keyboard shortcuts |\\n| Writing | Review paste and undo |\\n';
let generation = 0;
let source = false;
addEventListener('message', event => {
  if(event.source !== parent || event.data?.type !== 'inkkit-demo-theme') return;
  const scheme = event.data.dark ? 'dark' : 'light';
  document.documentElement.style.colorScheme = scheme;
  root.style.setProperty('--inkkit-color-scheme', scheme);
});
try {
  const { InkKitEditor } = await import('https://cdn.jsdelivr.net/npm/@aicayzer/inkkit@0.1.7/dist/core.js');
  const editor = await InkKitEditor.mount(root, {
    openLink() {},
    changed() { status.textContent = 'Your changes stay in this demo and are not saved.'; },
    error(error) { status.textContent = 'That edit could not be applied (' + error.code + '). You can keep editing.'; }
  }, { keymap: { bold: ['Mod-b'], italic: ['Mod-i'] } });
  const load = () => {
    editor.loadDocument({ documentId: 'demo', generation: ++generation, format: 'md', text: initial });
    source = false;
    mode.setAttribute('aria-pressed', 'false');
    status.textContent = 'Your changes stay in this demo and are not saved.';
  };
  load();
  mode.disabled = reset.disabled = false;
  mode.onclick = () => {
    try {
      editor.setEditingMode(source ? 'formatted' : 'source', generation);
      source = !source;
      mode.setAttribute('aria-pressed', String(source));
    } catch { status.textContent = 'Could not switch views. Your text is still here.'; }
  };
  reset.onclick = () => { try { load(); } catch { status.textContent = 'Could not reset yet. Finish the current edit and try again.'; } };
} catch (error) {
  console.error(error);
  status.textContent = 'The demo could not load. Refresh the page to try again.';
}
</script></body></html>`;
  return (
    <iframe
      ref={frame}
      title="Try the Inkkit editor"
      srcDoc={html}
      sandbox="allow-scripts"
      onLoad={sendTheme}
      className={fullPage ? "inkkit-demo inkkit-demo-full" : "inkkit-demo"}
    />
  );
};
