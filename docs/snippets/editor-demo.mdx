export const EditorDemo = ({
  fullPage = false,
  focused = false,
  onFocusChange,
}) => {
  const frame = useRef(null)
  const [url, setUrl] = useState(null)
  useEffect(() => {
    setUrl(
      `https://aicayzer.github.io/inkkit/${fullPage ? 'full' : 'compact'}.html?visit=${Date.now()}`,
    )
  }, [fullPage])
  const post = (message) =>
    frame.current?.contentWindow?.postMessage(message, '*')
  const sendTheme = () => {
    const navbarLogo = document
      .querySelector('#navbar .nav-logo')
      ?.closest('a')
      ?.getBoundingClientRect()
    const navbarLine = document.querySelector('#navbar-transition')
    let element = frame.current?.parentElement
    let background
    while (element) {
      const colour = getComputedStyle(element).backgroundColor
      if (colour !== 'transparent' && colour !== 'rgba(0, 0, 0, 0)') {
        background = colour
        break
      }
      element = element.parentElement
    }
    if (background)
      frame.current
        ?.closest('.editor-page')
        ?.style.setProperty('--demo-background', background)
    post({
      type: 'inkkit-demo-theme',
      dark: document.documentElement.classList.contains('dark'),
      background,
      divider: navbarLine
        ? getComputedStyle(navbarLine).borderBottomColor
        : undefined,
      navbarInset: navbarLogo?.width ? navbarLogo.left : undefined,
    })
  }
  useEffect(() => {
    const observer = new MutationObserver(sendTheme)
    observer.observe(document.documentElement, {
      attributes: true,
      attributeFilter: ['class'],
    })
    const navbarLine = document.querySelector('#navbar-transition')
    navbarLine?.addEventListener('transitionend', sendTheme)
    window.addEventListener('resize', sendTheme)
    return () => {
      observer.disconnect()
      navbarLine?.removeEventListener('transitionend', sendTheme)
      window.removeEventListener('resize', sendTheme)
    }
  }, [])
  useEffect(() => {
    post({ type: 'inkkit-demo-state', focused })
  }, [focused])
  useEffect(() => {
    const onMessage = (event) => {
      if (event.source !== frame.current?.contentWindow) return
      if (event.data?.type === 'inkkit-demo-focus') {
        onFocusChange?.(event.data.focused === true)
      } else if (fullPage && event.data?.type === 'inkkit-demo-height') {
        const height = event.data.height
        if (Number.isFinite(height) && height >= 410) {
          frame.current.style.height = Math.min(height, 100000) + 'px'
        }
      }
    }
    window.addEventListener('message', onMessage)
    return () => window.removeEventListener('message', onMessage)
  }, [onFocusChange, fullPage])
  return (
    <iframe
      ref={frame}
      title="Try the Inkkit editor"
      src={url ?? undefined}
      sandbox="allow-scripts allow-popups allow-popups-to-escape-sandbox"
      allow="clipboard-write"
      onLoad={() => {
        sendTheme()
        post({ type: 'inkkit-demo-state', focused })
      }}
      className={fullPage ? 'inkkit-demo inkkit-demo-full' : 'inkkit-demo'}
    />
  )
}

export const EditorPage = () => {
  const [focused, setFocused] = useState(false)
  useEffect(() => {
    document.documentElement.classList.toggle('editor-focus', focused)
    return () => document.documentElement.classList.remove('editor-focus')
  }, [focused])
  useEffect(() => {
    if (!focused) return
    const onKeyDown = (event) => {
      if (event.key === 'Escape' && !event.defaultPrevented) setFocused(false)
    }
    document.addEventListener('keydown', onKeyDown)
    return () => document.removeEventListener('keydown', onKeyDown)
  }, [focused])
  return (
    <div className="editor-page">
      <EditorDemo fullPage focused={focused} onFocusChange={setFocused} />
    </div>
  )
}
